class Node:
    def __init__(self, data):
        self.data = data
        self.left = None
        self.right = None


# INSERTION
def insert(root, value):
    if root is None:
        return Node(value)

    if value < root.data:
        root.left = insert(root.left, value)

    elif value > root.data:
        root.right = insert(root.right, value)

    return root


# SEARCHING
def search(root, key):
    if root is None:
        return False

    if root.data == key:
        return True

    elif key < root.data:
        return search(root.left, key)

    else:
        return search(root.right, key)


# DISPLAY
def display(root):
    if root is not None:
        display(root.left)
        print(root.data, end=" ")
        display(root.right)


# DELETION
def delete(root, key):
    if root is None:
        return None

    if key < root.data:
        root.left = delete(root.left, key)

    elif key > root.data:
        root.right = delete(root.right, key)

    else:
        if root.left is None:
            return root.right

        elif root.right is None:
            return root.left

        temp = root.right

        while temp.left is not None:
            temp = temp.left

        root.data = temp.data
        root.right = delete(root.right, temp.data)

    return root


# MAIN PROGRAM
root = None

while True:
    print("\n1. Insert")
    print("2. Delete")
    print("3. Display")
    print("4. Search")
    print("5. Exit")

    ch = int(input("Enter choice: "))

    if ch == 1:
        value = int(input("Enter value: "))
        root = insert(root, value)

    elif ch == 2:
        value = int(input("Enter value to delete: "))
        root = delete(root, value)

    elif ch == 3:
        display(root)
        print()

    elif ch == 4:
        value = int(input("Enter value to search: "))

        if search(root, value):
            print("Value found")
        else:
            print("Value not found")

    elif ch == 5:
        break

    else:
        print("Invalid choice")
