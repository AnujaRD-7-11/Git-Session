undo_stack = []
redo_stack = []
document = ""

while True:
    print("\n1. Make a Change")
    print("2. Undo")
    print("3. Redo")
    print("4. Display Document")
    print("5. Exit")

    choice = int(input("Enter choice: "))

    if choice == 1:
        undo_stack.append(document)
        change = input("Enter text: ")
        document = document + change
        redo_stack.clear()

    elif choice == 2:
        if len(undo_stack) == 0:
            print("Nothing to undo")
        else:
            redo_stack.append(document)
            document = undo_stack.pop()
            print("Undo successful")

    elif choice == 3:
        if len(redo_stack) == 0:
            print("Nothing to redo")
        else:
            undo_stack.append(document)
            document = redo_stack.pop()
            print("Redo successful")

    elif choice == 4:
        print("Current Document:", document)
        print("Undo Stack:", undo_stack)
        print("Redo Stack:", redo_stack)

    elif choice == 5:
        print("Exiting...")
        break

    else:
        print("Invalid choice")
